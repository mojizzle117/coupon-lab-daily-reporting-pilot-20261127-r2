class JoinError(Exception):
    pass


def sha256(content):
    return hashlib.sha256(content).hexdigest()


def canonical_part(directory, name, expected_hash, tail):
    path = directory / name
    if not path.is_file():
        raise JoinError(name + ": missing. Open its Raw link, copy all text into a separate UTF-8 file with this exact name, and retry.")
    try:
        text = path.read_bytes().decode("utf-8-sig")
    except (OSError, UnicodeError):
        raise JoinError(name + ": cannot read UTF-8 text. Repaste that part into a plain-text UTF-8 file and retry.") from None
    text = text.replace("\r\n", "\n").replace("\r", "\n")
    core = text.rstrip(" \t\n")
    content = (core + tail).encode("utf-8")
    if sha256(content) != expected_hash:
        for other_name, other_hash, other_tail in PARTS:
            if other_name != name and sha256((core + other_tail).encode("utf-8")) == other_hash:
                raise JoinError(name + ": contains " + other_name + " instead. Repaste the correct " + name + " from its Raw link and retry.")
        raise JoinError(name + ": incomplete or changed. Repaste the entire correct part from its Raw link into this file and retry.")
    return content


def existing_output_matches(path):
    try:
        if sha256(path.read_bytes()) == FULL_SHA256:
            print("Already verified COPY_ME_FIRST.py.txt; kept unchanged.")
            return True
    except OSError:
        pass
    raise JoinError("COPY_ME_FIRST.py.txt: an existing file differs or cannot be read. It was kept unchanged. Keep your copy and assemble this kit in a fresh folder.")


def main():
    directory = Path(__file__).resolve().parent
    assembled = b"".join(canonical_part(directory, name, digest, tail)
                         for name, digest, tail in PARTS)
    if sha256(assembled) != FULL_SHA256:
        raise JoinError("Complete installer hash differs. Repaste JOIN_PARTS.py.txt and all 20 correctly named parts from the same kit, then retry in a fresh folder.")
    output = directory / "COPY_ME_FIRST.py.txt"
    if output.exists():
        existing_output_matches(output)
    else:
        created = False
        try:
            with output.open("xb") as handle:
                created = True
                handle.write(assembled)
        except FileExistsError:
            existing_output_matches(output)
        except OSError:
            if created:
                try:
                    output.unlink()
                except OSError:
                    pass
            raise JoinError("COPY_ME_FIRST.py.txt: could not save the complete installer. Check folder write permission and retry in a fresh writable folder.") from None
        else:
            print("Created and verified COPY_ME_FIRST.py.txt from all 20 parts.")
    print("Next, run in this folder:")
    print("python COPY_ME_FIRST.py.txt --guide")
    return 0


if __name__ == "__main__":
    try:
        raise SystemExit(main())
    except JoinError as error:
        print("Assembly stopped. " + str(error), file=sys.stderr)
        raise SystemExit(1)
