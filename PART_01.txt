PAYLOAD_SHA256 = "3dd86745b05617cd066fca73913cd055aea13394514a5e57726d6180196ff42d"
COPY_ERROR = ("The pasted source is incomplete or changed. Paste the entire text into a "
              "blank file again, from the first line through '# END OFFICE PAYLOAD', then retry.")


def read_payload():
    source = Path(__file__).read_text(encoding="utf-8-sig").split("\n")
    try:
        start = source.index("# BEGIN OFFICE PAYLOAD")
        end = source.index("# END OFFICE PAYLOAD", start + 1)
    except ValueError:
        raise ValueError(COPY_ERROR) from None
    if any(line.strip() for line in source[end + 1:]):
        raise ValueError(COPY_ERROR)
    kit = None
    companions = {}
    current = None
    older, newer = [], []

    def finish_record():
        if current is None:
            return
        new_text = "\n".join(newer) + ("\n" if current["new_lf"] else "")
        name = current["name"]
        if current["kind"] == "companion":
            if name in companions:
                raise ValueError(COPY_ERROR)
            companions[name] = new_text
        elif current["kind"] == "app" and name not in kit["files"]:
            old_text = None
            if "old_lf" in current:
                old_text = "\n".join(older) + ("\n" if current["old_lf"] else "")
            kit["files"][name] = {"base": old_text, "new": new_text}
        else:
            raise ValueError(COPY_ERROR)

    for line in source[start + 1:end]:
        if line.startswith("# COPY BLOCK "):
            continue
        if line.startswith("# METADATA ") and kit is None:
            kit = json.loads(line.removeprefix("# METADATA "))
            kit["files"] = {}
        elif line.startswith("# FILE ") and kit is not None:
            finish_record()
            current = json.loads(line.removeprefix("# FILE "))
            older, newer = [], []
        elif current is not None and line[:2] in ("#=", "#-", "#+"):
            if line[1] in "=-":
                older.append(line[2:])
            if line[1] in "=+":
                newer.append(line[2:])
        else:
            raise ValueError(COPY_ERROR)
    finish_record()
    canonical = json.dumps([kit, companions], ensure_ascii=False, sort_keys=True, separators=(",", ":"))
    if hashlib.sha256(canonical.encode("utf-8")).hexdigest() != PAYLOAD_SHA256:
        raise ValueError(COPY_ERROR)
    return kit, companions


def write_guides(companions):
    directory = Path(__file__).resolve().parent
    for name, value in companions.items():
        path = directory / name
        try:
            with path.open("x", encoding="utf-8") as handle:
                handle.write(value)
        except FileExistsError:
            if path.read_text(encoding="utf-8-sig") != value:
                print(f"Kept your changed {name}. To recover the supplied version, paste this installer into a new folder and run --guide there.")
            else:
                print(f"Already present: {name}")
        else:
            print(f"Created {name}")
    print(f"Open FRIDAY_OFFICE_GUIDE.txt in {directory}")


def body(raw):
    return raw.decode("utf-8-sig").replace("\r\n", "\n")


def encode_like(value, original):
    if b"\r\n" in original:
        value = value.replace("\n", "\r\n")
    codec = "utf-8-sig" if original.startswith(b"\xef\xbb\xbf") else "utf-8"
    return value.encode(codec)


def merge_code(path, current, base, proposed):
    if current == proposed:
        return current
    if current == base:
        return proposed
    if base is None or current is None:
        raise ValueError(f"{path}: file differs or is missing. Use --export to inspect the update.")
    if not shutil.which("git"):
        raise ValueError(f"{path}: office edits detected. Use local Git for the merge, or --export for a manual merge.")
    with tempfile.TemporaryDirectory() as temp:
        files = [Path(temp) / key for key in ("office", "baseline", "pilot")]
        for dest, content in zip(files, (current, base, proposed)):
            dest.write_bytes(content.encode("utf-8"))
        result = subprocess.run(["git", "merge-file", "-p", *files], capture_output=True)
    if result.returncode:
        raise ValueError(f"{path}: overlapping office edits. Use --export, merge the patch in your office copy, and rerun --check.")
    return result.stdout.decode("utf-8")


def merge_products(raw, additions, platform_a):
    reader = csv.DictReader(io.StringIO(body(raw)))
    rows = list(reader)
    fields = reader.fieldnames
    if not fields or any(key not in fields for key in additions[0]):
        raise ValueError("Product mapping columns differ. Review the additions in the payload.")
    extra = io.StringIO()
    writer = csv.DictWriter(extra, fields, lineterminator="\r\n" if b"\r\n" in raw else "\n")
    for row in additions:
        matches = [old for old in rows if old["Platform"] in platform_a and old["Source_product"] == row["Source_product"]]
        if len(matches) == 1 and all(matches[0][key].strip().casefold() == row[key].strip().casefold()
                                     for key in ("Common_product", "Barrier", "Capped", "Reviewed")):
            continue
        if matches:
            raise ValueError("A target product mapping is duplicated or has different attributes. Reconcile the two target labels before applying.")
        # Preserve the office's platform spelling when its reviewed reference
        # uses a private brand assignment. Otherwise use the canonical slot.
        spelling = next((old["Platform"] for old in rows if old["Platform"] in platform_a), row["Platform"])
        writer.writerow({**row, "Platform": spelling})
    if not extra.tell():
        return raw
    return raw + (b"" if raw.endswith(b"\n") else b"\n") + extra.getvalue().encode("utf-8")


def checked_target(app, relative):
    path = (app / relative).resolve()
    if not path.is_relative_to(app):
        raise ValueError("A target path lies outside the office copy. Copy its reference inside the pilot folder and update its setting.")
    return path


def main():
    parser = argparse.ArgumentParser(description="Coupon Lab copy-paste office pilot r2")
    parser.add_argument("--app", type=Path, default=Path.cwd())
    modes = parser.add_mutually_exclusive_group(required=True)
    modes.add_argument("--guide", action="store_true", help="write the guide, office checks and return sheet beside this file")
    modes.add_argument("--check", action="store_true", help="check the pilot app copy without changing it")
    modes.add_argument("--apply", action="store_true", help="apply to the pilot app copy, with a backup")
    modes.add_argument("--export", type=Path, metavar="NEW_FOLDER", help="write patch references for an office manual merge")
    args = parser.parse_args()
    kit, companions = read_payload()
    if args.guide:
        write_guides(companions)
        return
    if args.export:
        args.export.mkdir(parents=True, exist_ok=False)
        for name, change in kit["files"].items():
            target = checked_target(args.export.resolve(), name)
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_bytes(change["new"].encode("utf-8"))
            if change["base"] is not None:
                patch = difflib.unified_diff(change["base"].splitlines(True), change["new"].splitlines(True), name, name)
                target.with_name(target.name + ".patch").write_text("".join(patch), encoding="utf-8")
        print("Exported pilot files and changes for an internal manual merge.")
        return
    app = args.app.resolve()
    if not (app / "run_coupon_lab.py").is_file() or not (app / "settings.json").is_file():
        raise ValueError("Pass --app pointing to the COPY containing run_coupon_lab.py and settings.json.")
    changes = {}
    for name, change in kit["files"].items():
        path = checked_target(app, name)
        raw = path.read_bytes() if path.exists() else None
        current = body(raw) if raw is not None else None
        merged = merge_code(name, current, change["base"], change["new"])
        if merged != current:
            changes[path] = (raw, encode_like(merged, raw if raw is not None else b""))
    settings = checked_target(app, "settings.json")
    raw = settings.read_bytes()
    cfg = json.loads(raw.decode("utf-8-sig"))
    daily = cfg.setdefault("daily_reporting", {})
    if not isinstance(daily, dict):
        raise ValueError("Expected daily_reporting to be an object in settings.json.")
    for key, value in kit["settings"].items():
        daily.setdefault(key, value)
    if cfg != json.loads(raw.decode("utf-8-sig")):
        changes[settings] = (raw, encode_like(json.dumps(cfg, ensure_ascii=False, indent=2) + "\n", raw))
    mapping = checked_target(app, cfg.get("references", {}).get("products", "reference/product_mappings.csv"))
    raw = mapping.read_bytes()
    platform_a = {"PLATFORM_A"}
    registry = app / cfg.get("brand_repository", "brand_repository.json")
    if registry.is_file():
        brand = json.loads(registry.read_text(encoding="utf-8-sig"))["brands"]["platform_a"]
        platform_a.update([brand.get("name"), brand.get("source_code"), *brand.get("source_aliases", [])])
    mapped = merge_products(raw, kit["products"], platform_a)
    if mapped != raw:
        changes[mapping] = (raw, mapped)
    print(f"{kit['version']} | {len(changes)} files to update.")
    for path in changes:
        print("  " + str(path.relative_to(app)))
    if not args.apply or not changes:
        return
    backup = Path(tempfile.mkdtemp(prefix="coupon_daily_backup_" + datetime.now().strftime("%Y%m%d") + "_", dir=app.parent))
    for path, (raw, _) in changes.items():
        if raw is not None:
            dest = backup / path.relative_to(app)
            dest.parent.mkdir(parents=True, exist_ok=True)
            dest.write_bytes(raw)
    written = []
    try:
        for path, (before, after) in changes.items():
            if (path.read_bytes() if path.exists() else None) != before:
                raise ValueError("Another process changed a target after the check. Close the pilot copy and try again.")
            path.parent.mkdir(parents=True, exist_ok=True)
            with tempfile.NamedTemporaryFile(dir=path.parent, delete=False) as staged:
                staged.write(after)
            try:
                os.replace(staged.name, path)
            finally:
                Path(staged.name).unlink(missing_ok=True)
            written.append(path)
    except Exception:
        for path in reversed(written):
            before = changes[path][0]
            if before is None:
                path.unlink()
            else:
                path.write_bytes(before)
        raise
    print(f"Installed. Originals are in the sibling folder {backup.name}. Follow FRIDAY_OFFICE_GUIDE.txt.")


if __name__ == "__main__":
    try:
        main()
    except (OSError, ValueError, KeyError, TypeError) as error:
        print(f"Pilot installation stopped. {error}", file=sys.stderr)
        sys.exit(1)
